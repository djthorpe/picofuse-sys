cmake_minimum_required(VERSION 3.16)

project(picofuse_example_sys_helloworld C)

find_package(PkgConfig REQUIRED)

set(PICOFUSE_PREFIX "/opt/picofuse" CACHE PATH "Installed picofuse prefix")
set(PICOFUSE_BOARD "pico" CACHE STRING "Installed picofuse board package to use")

set(_PICOFUSE_PKG_PATHS
    "${PICOFUSE_PREFIX}/lib/pkgconfig"
    "${PICOFUSE_PREFIX}/lib64/pkgconfig"
    "${PICOFUSE_PREFIX}/share/pkgconfig"
)
list(JOIN _PICOFUSE_PKG_PATHS ":" _PICOFUSE_PKG_PATH)
if(DEFINED ENV{PKG_CONFIG_PATH} AND NOT "$ENV{PKG_CONFIG_PATH}" STREQUAL "")
    set(ENV{PKG_CONFIG_PATH} "${_PICOFUSE_PKG_PATH}:$ENV{PKG_CONFIG_PATH}")
else()
    set(ENV{PKG_CONFIG_PATH} "${_PICOFUSE_PKG_PATH}")
endif()

set(PICOFUSE_PKG_NAME "picofuse-${PICOFUSE_BOARD}")
pkg_check_modules(PICOFUSE REQUIRED ${PICOFUSE_PKG_NAME})
pkg_get_variable(PICOFUSE_PACKAGE_PREFIX ${PICOFUSE_PKG_NAME} prefix)
pkg_get_variable(PICOFUSE_ARMGNU_VALUE ${PICOFUSE_PKG_NAME} PICOFUSE_ARMGNU)
pkg_get_variable(PICOFUSE_CPU_VALUE ${PICOFUSE_PKG_NAME} PICOFUSE_CPU)
pkg_get_variable(PICOFUSE_CHIP_DIR_VALUE ${PICOFUSE_PKG_NAME} PICOFUSE_CHIP_DIR)
pkg_get_variable(PICOFUSE_FAMILY_VALUE ${PICOFUSE_PKG_NAME} PICOFUSE_FAMILY)
pkg_get_variable(PICOFUSE_SRAM_ENTRY_VALUE ${PICOFUSE_PKG_NAME} PICOFUSE_SRAM_ENTRY)
pkg_get_variable(PICOFUSE_SRAM_LOAD_VALUE ${PICOFUSE_PKG_NAME} PICOFUSE_SRAM_LOAD)
pkg_get_variable(PICOFUSE_FLASH_ENTRY_VALUE ${PICOFUSE_PKG_NAME} PICOFUSE_FLASH_ENTRY)
pkg_get_variable(PICOFUSE_FLASH_LOAD_VALUE ${PICOFUSE_PKG_NAME} PICOFUSE_FLASH_LOAD)

execute_process(
    COMMAND ${PKG_CONFIG_EXECUTABLE} --libs-only-l ${PICOFUSE_PKG_NAME}
    OUTPUT_VARIABLE PICOFUSE_LIBRARIES_RAW
    OUTPUT_STRIP_TRAILING_WHITESPACE
)
execute_process(
    COMMAND ${PKG_CONFIG_EXECUTABLE} --libs-only-other ${PICOFUSE_PKG_NAME}
    OUTPUT_VARIABLE PICOFUSE_LINK_OPTIONS_RAW
    OUTPUT_STRIP_TRAILING_WHITESPACE
)
separate_arguments(PICOFUSE_LIBRARIES_RAW UNIX_COMMAND "${PICOFUSE_LIBRARIES_RAW}")
separate_arguments(PICOFUSE_LINK_OPTIONS_RAW UNIX_COMMAND "${PICOFUSE_LINK_OPTIONS_RAW}")

set(_PICOFUSE_REPO_CMAKE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../cmake")
if(EXISTS "${_PICOFUSE_REPO_CMAKE_DIR}/picofuse.cmake")
    list(PREPEND CMAKE_MODULE_PATH "${_PICOFUSE_REPO_CMAKE_DIR}")
endif()
list(APPEND CMAKE_MODULE_PATH "${PICOFUSE_PACKAGE_PREFIX}/share/picofuse/cmake")
include(picofuse)

set(PICOFUSE_INCLUDE_DIRS ${PICOFUSE_INCLUDE_DIRS})
set(PICOFUSE_LIBRARY_DIRS ${PICOFUSE_LIBRARY_DIRS})
set(PICOFUSE_LIBRARIES ${PICOFUSE_LIBRARIES_RAW})
set(PICOFUSE_LINK_OPTIONS ${PICOFUSE_LINK_OPTIONS_RAW})
set(PICOFUSE_CPU ${PICOFUSE_CPU_VALUE})
set(PICOFUSE_ARMGNU ${PICOFUSE_ARMGNU_VALUE})
set(PICOFUSE_CHIP_DIR ${PICOFUSE_CHIP_DIR_VALUE})
set(PICOFUSE_FAMILY ${PICOFUSE_FAMILY_VALUE})
set(PICOFUSE_SRAM_ENTRY ${PICOFUSE_SRAM_ENTRY_VALUE})
set(PICOFUSE_SRAM_LOAD ${PICOFUSE_SRAM_LOAD_VALUE})
set(PICOFUSE_FLASH_ENTRY ${PICOFUSE_FLASH_ENTRY_VALUE})
set(PICOFUSE_FLASH_LOAD ${PICOFUSE_FLASH_LOAD_VALUE})

message(STATUS "Using ${PICOFUSE_PKG_NAME} from ${PICOFUSE_PACKAGE_PREFIX}")

picofuse_add_executable(
    NAME hello-world
    C main.c
    LDSCRIPT sram.ld
)
